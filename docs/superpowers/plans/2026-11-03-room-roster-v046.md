**Goal:** Archive every message of a room whoever posted it. This covers authors outside the API member list, messages without `member_id`, and canceled messages. A 429 must never look like "no data".

**Architecture:**
- A pure `pysaka.roster` module derives authors from messages.
- `SyncManager` gains:

  - history-scan helpers (`history_scanned`, `mark_history_scanned`, `room_archive_complete`);
  - `sync_member(ignore_cursor=)`;
  - member-0 (room author) handling;
  - folder renames.
- `Client` gains `get_member`, `get_messages(page_delay=)` and `RateLimitedError`.
- `_merge_messages` keeps captured content when a message becomes canceled.
- saka-cli wires the new API into its Phase 2.

**Tech Stack:** Python 3.9+, asyncio, aiohttp, structlog, pytest (asyncio auto mode), time-machine; uv only.

Spec: `docs/superpowers/specs/2026-11-03-room-roster-v046-design.md` (read §1–§10 once).

```python
class RateLimitedError(ApiError):
    """The API answered 429 Too Many Requests.

    Raised instead of returning None so a rate-limited response is never mistaken
    for "no data" (an empty group list, timeline or member list). Retry later.

    Attributes:
        retry_after: Seconds from the ``Retry-After`` header, or None when absent
            or not a number.
    """

    def __init__(self, message: str, retry_after: Optional[float] = None):
        super().__init__(message, 429)
        self.retry_after = retry_after
```

```python
"""Who authored the messages of one room.

``GET /groups/{id}/members`` returns a room's *current* lineup, not everyone who
ever posted there: rotating-lineup rooms (one room reused for many live events)
and official rooms (only the official account is listed) hold messages from
members missing from that list. These helpers derive the authors from the
messages themselves. No I/O.
"""

from collections.abc import Iterable
from typing import Any

ROOM_AUTHOR_ID = 0
"""Member id for messages the API returns without ``member_id``. The official app
shows those as the room itself (room name and logo)."""


def author_id(message: dict[str, Any]) -> int:
    """The author's member id, or ``ROOM_AUTHOR_ID`` when the message has none."""
    mid = message.get("member_id")
    if isinstance(mid, int) and not isinstance(mid, bool):
        return mid
    if isinstance(mid, str) and mid.isdigit():
        return int(mid)
    return ROOM_AUTHOR_ID


def extra_author_ids(api_member_ids: Iterable[int], messages: Iterable[dict[str, Any]]) -> list[int]:
    """Authors of ``messages`` missing from ``api_member_ids``.

    Real member ids come first in ascending order; ``ROOM_AUTHOR_ID`` comes last
    when any message has no author.
    """
    known = set(api_member_ids)
    authors = {author_id(m) for m in messages}
    real = sorted(a for a in authors if a != ROOM_AUTHOR_ID and a not in known)
    room = [ROOM_AUTHOR_ID] if ROOM_AUTHOR_ID in authors and ROOM_AUTHOR_ID not in known else []
    return real + room


def room_author(group: dict[str, Any]) -> dict[str, Any]:
    """The member dict that stands for the room itself (member 0)."""
    return {"id": ROOM_AUTHOR_ID, "name": group.get("name") or str(group["id"]), "thumbnail": group.get("thumbnail")}
```

```python
    def _ensure_member_dir(self, group_dir: Path, member_id: int, name: str) -> Path:
        """Return ``group_dir/<mid> <name>``, renaming a member folder left under an old name.

        A member who changed their display name, or an author first archived under
        a fallback name (their id), keeps one folder instead of a duplicate. Only
        renames when the target does not exist yet; never creates the directory.
        """
        member_dir = group_dir / f"{member_id} {name}"
        if not member_dir.exists() and group_dir.exists():
            prefix = f"{member_id} "
            for existing in group_dir.iterdir():
                if existing.is_dir() and existing.name.startswith(prefix):
                    logger.info("Member renamed, renaming directory", member_id=member_id, old=existing.name, new=member_dir.name)
                    existing.rename(member_dir)
                    break
        return member_dir
```

```python
    _MEMBER_PROFILE_KEYS = ("thumbnail", "phone_image", "portrait")

    async def extra_room_members(
        self,
        session: aiohttp.ClientSession,
        group: dict[str, Any],
        api_members: list[dict[str, Any]],
        messages: list[dict[str, Any]],
        name_hints: Optional[dict[int, dict[str, Any]]] = None,
    ) -> list[dict[str, Any]]:
        """Members to sync for this room **in addition to** ``api_members``.

        ``/groups/{id}/members`` is the room's current lineup; earlier lineups and
        members posting in official rooms are missing from it, and sync would drop
        their messages. This returns one member dict per author of ``messages``
        not in ``api_members`` (ascending id), then member 0 (the room author,
        ``pysaka.roster.room_author``) when a message has no ``member_id``.

        A name is resolved from, in order: ``name_hints`` (id -> member dict, e.g.
        every room's member list fetched in this sync); the member block of a local
        ``messages/* */<id> */messages.json`` (unless it only holds the fallback
        name); ``Client.get_member`` (cached per manager); else ``str(id)``. A
        message is never dropped because its author's name is unknown; a later sync
        that learns the name renames the folder.

        Raises:
            ApiError: From ``Client.get_member`` (including ``RateLimitedError``).
        """
        hints = name_hints or {}
        result: list[dict[str, Any]] = []
        for aid in roster.extra_author_ids((m["id"] for m in api_members), messages):
            if aid == roster.ROOM_AUTHOR_ID:
                result.append(roster.room_author(group))
            else:
                result.append(await self._resolve_member(session, aid, hints))
        return result

    async def _resolve_member(
        self, session: aiohttp.ClientSession, member_id: int, hints: dict[int, dict[str, Any]]
    ) -> dict[str, Any]:
        hinted = hints.get(member_id)
        if hinted and hinted.get("name"):
            return dict(hinted)
        local = await asyncio.to_thread(self._local_member_block, member_id)
        if local:
            return local
        if member_id not in self._member_cache:
            fetched = await self.client.get_member(session, member_id)
            self._member_cache[member_id] = fetched if fetched and fetched.get("name") else None
        cached = self._member_cache[member_id]
        if cached:
            return {"id": member_id, "name": cached["name"], **{k: cached.get(k) for k in self._MEMBER_PROFILE_KEYS}}
        return {"id": member_id, "name": str(member_id)}

    def _local_member_block(self, member_id: int) -> Optional[dict[str, Any]]:
        """Member block of any archived room folder for ``member_id`` with a real name."""
        messages_dir = self.output_dir / "messages"
        if not messages_dir.exists():
            return None
        for path in messages_dir.glob(f"* */{member_id} */messages.json"):
            try:
                with open(path, encoding="utf-8") as f:
                    block = json.load(f).get("member")
            except (OSError, ValueError, AttributeError):
                continue
            if isinstance(block, dict) and block.get("name") and block["name"] != str(member_id):
                return {"id": member_id, "name": block["name"], **{k: block.get(k) for k in self._MEMBER_PROFILE_KEYS}}
        return None
```

```python
    @staticmethod
    def _room_state_key(group_id: int) -> str:
        # Never collides with "<gid>_<mid>" member keys.
        return f"room_{group_id}"

    def history_scanned(self, group_id: int) -> bool:
        """True once the room's one-time full-history scan has completed.

        Stored as ``sync_state["room_<gid>"]["history_scanned_at"]``; deleting
        ``sync_state.json`` (a force resync) clears it, so the next sync scans again.
        """
        return bool((self.sync_state.get(self._room_state_key(group_id)) or {}).get("history_scanned_at"))

    def mark_history_scanned(self, group_id: int) -> None:
        """Record that the room's full history is archived (call only after
        :meth:`room_archive_complete` returned True for the scan's timeline)."""
        key = self._room_state_key(group_id)
        entry = dict(self.sync_state.get(key) or {})
        entry["history_scanned_at"] = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
        self.sync_state[key] = entry
        self.save_sync_state()

    async def room_archive_complete(self, group: dict[str, Any], messages: list[dict[str, Any]]) -> bool:
        """True when every message id in ``messages`` is in its author's ``messages.json``.

        Authors are taken with :func:`pysaka.roster.author_id` (member 0 for messages
        without ``member_id``). False on an unreadable file. Catches what a raised
        exception would not: a write that failed (``sync_member`` returns 0) or a
        message that failed to normalize.
        """
        return await asyncio.to_thread(self._room_archive_complete_sync, group, messages)

    def _room_archive_complete_sync(self, group: dict[str, Any], messages: list[dict[str, Any]]) -> bool:
        wanted: dict[int, set[int]] = {}
        for m in messages:
            wanted.setdefault(roster.author_id(m), set()).add(m["id"])
        messages_dir = self.output_dir / "messages"
        group_dirs = list(messages_dir.glob(f"{group['id']} *")) if messages_dir.exists() else []
        for aid, ids in wanted.items():
            stored: set[Any] = set()
            for group_dir in group_dirs:
                for path in group_dir.glob(f"{aid} */messages.json"):
                    try:
                        with open(path, encoding="utf-8") as f:
                            data = json.load(f)
                    except (OSError, ValueError):
                        return False
                    if not isinstance(data, dict):
                        return False
                    stored |= {m.get("id") for m in data.get("messages", []) if isinstance(m, dict)}
            if not ids <= stored:
                logger.warning("Room archive incomplete", group_id=group["id"], member_id=aid, missing=len(ids - stored))
                return False
        return True
```
